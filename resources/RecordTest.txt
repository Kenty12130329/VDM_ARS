-- 正しいコード

class RecordTest

types
    public Token = token;
    public book :: 
        name : Token
        borrower_name : [Token];

    public Person ::
        name : Token
        books : set of Token;

instance variables
    public borrowable_books : set of book := {mk_book(mk_token("book_A"), nil), mk_book(mk_token("book_B"), nil)};
    public group : set of Person := {mk_Person(mk_token("user1"), {})};
    
    inv forall p in set group & card(p.books) <= 1;

operations
    public addBook: () ==> ()
    addBook() == (
        dcl b : book;
        dcl p : Person;
        
        b := let x in set borrowable_books in x;
        p := let y in set group be st card(y.books) == 0 in y;
        
        borrowable_books := borrowable_books \ {b};
        group := group \ {p};
        
        -- 本を追加
        p.books := p.books union {b.name};
        
        group := group union {p};
    )
    pre borrowable_books <> {};

    public returnBook: () ==> ()
    returnBook() == (
        dcl p : Person;
        dcl t : Token;
        dcl b : book;
        
        p := let y in set group be st card(y.books) > 0 in y;
        t := let x in set p.books in x;
        
        group := group \ {p};
        
        -- 本を返却
        p.books := p.books \ {t};
        
        group := group union {p};
        
        -- 在庫に戻す
        b := mk_book(t, nil);
        borrowable_books := borrowable_books union {b};
    )
    pre exists p in set group & card(p.books) > 0;

end RecordTest

class RecordTest
types
    -- alla.vdmpp と同一の定義
    public Token = token;
    
    public book :: 
        book_name : Token
        borrower_name : [Token];

    public Person :: 
        name : Token
        books : set of Token;

instance variables
    -- 在庫：book_A と book_B の2冊を用意
    public borrowable_books : set of book := {
        mk_book(mk_token("book_A"), nil), 
        mk_book(mk_token("book_B"), nil)
    };
    
    public group : set of Person := {
        mk_Person(mk_token("user1"), {})
    };
    
    -- 不変条件：一人1冊まで
    inv forall p in set group & card(p.books) <= 1;

operations

    public addBook: () ==> ()
    addBook() == (
        -- 1. 宣言 (Declaration)
        dcl b : book;
        dcl p : Person;
        
        -- 2. 在庫から本を1つ選択
        b := let x in set borrowable_books in x;
        
        -- 3. [BUG INJECTION] ユーザーを選択 (借りている数に関係なく誰でも選べるように変更)
        -- 正しいガード: be st card(y.books) == 0
        p := let y in set group in y;
        
        -- 4. 在庫とユーザーリストから一旦削除
        borrowable_books := borrowable_books \ {b};
        group := group \ {p};
        
        -- 5. 本を追加 (ここで2冊目以降も追加されてしまう)
        p.books := p.books union {b.name};
        
        -- 6. 更新したユーザーをリストに戻す
        group := group union {p};
    )
    pre borrowable_books <> {};

    public returnBook: () ==> ()
    returnBook() == (
        dcl p : Person;
        dcl t : Token;
        dcl b : book;
        
        p := let y in set group be st card(y.books) > 0 in y;
        t := let x in set p.books in x;
        
        group := group \ {p};
        
        -- 本を返却
        p.books := p.books \ {t};
        
        group := group union {p};
        
        -- 在庫に戻す
        b := mk_book(t, nil);
        borrowable_books := borrowable_books union {b};
    )
    pre exists p in set group & card(p.books) > 0;


end RecordTest
